*** READ THIS — percept-stack is the car ***
-------------------------------------
carla-nav2-avl percept-stack owns ZEDs/sensors on boot.
DO NOT run orin_launch_all.sh while percept is up — it kills ZEDs/Argus
and will break WebUI + perception.

WebUI says Disconnected (service still "active"):
  sudo journalctl -u avros-webui -n 50 --no-pager
  sudo systemctl restart avros-webui
  sleep 5
  systemctl is-active avros-webui
  ss -ltnp | grep 8000
  rem then hard-refresh https://100.93.121.3:8000

PREFERRED clean launch (H.265 + wheel, NO JPEG — avoids link meltdown):
  bash ~/Documents/SimChair_Teleop/simchair_launch_clean.sh
  # Windows CMD:  set VIDEO_MODE=zed  then  start_all.bat

Attach variants:
  bash ~/Documents/SimChair_Teleop/simchair_attach.sh --ros-drive --no-cam
  bash ~/Documents/SimChair_Teleop/simchair_attach.sh --ros-drive   # also starts JPEG — don't with H.265
  bash ~/Documents/SimChair_Teleop/simchair_attach.sh --drive       # LEGACY serial steal

  Kill ONLY us (leaves percept alone):
    tmux kill-session -t simchair
    sudo pkill -f orin_simchair_ros_bridge
    sudo pkill -f orin_vehicle_controller   # only if legacy --drive
    sudo systemctl restart avros-webui      # only if legacy --drive

  CAMERA_SOURCE=raw CAMERA_PROFILE=default  # 1280x720 — can crash Jetson if percept is loaded
  CAMERA_PROFILE=hq          # only with raw; heaviest
